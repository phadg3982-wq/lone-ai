Google Apps Script Code

Open a new Google Sheet → Click Extensions > Apps Script → Paste this code → Click Deploy > New Deployment > Web App (Execute as: Me, Who has access: Anyone) → Copy the Web App URL into Settings! 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", "Applicant Name", "Age", "Monthly Salary",
        "Credit Score", "Existing EMI", "Status",
        "Eligible Loan Amount", "Risk Rating", "DTI Ratio"
      ]);
      sheet.getRange(1, 1, 1, 10).setFontWeight("bold").setBackground("#e2e8f0");
    }
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.timestamp || new Date().toISOString(),
      data.applicantName || "N/A",
      data.age || "N/A",
      data.salary || 0,
      data.creditScore || 0,
      data.existingEmi || 0,
      data.status || "N/A",
      data.eligibleAmount || 0,
      data.riskClassification || "N/A",
      data.dtiRatio || "N/A"
    ]);
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
